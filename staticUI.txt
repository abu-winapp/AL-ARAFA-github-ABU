export default function GuestReviews() {
  const sectionRef = useRef(null);

  const previousRef = useRef(null);
  const activeRef = useRef(null);
  const nextRef = useRef(null);

  const [activeIndex, setActiveIndex] = useState(0);
  const [direction, setDirection] = useState(1);
  const [animating, setAnimating] = useState(false);

  const getIndex = (index) => {
    return (index + reviews.length) % reviews.length;
  };

  const previousIndex = getIndex(activeIndex - 1);
  const nextIndex = getIndex(activeIndex + 1);

  /*
   * INITIAL SCROLL ANIMATION
   */
  useLayoutEffect(() => {
    const ctx = gsap.context(() => {
      gsap.from(".review-heading", {
        y: 35,
        opacity: 0,
        duration: 1,
        ease: "power3.out",
      });

      gsap.from(".review-description", {
        y: 20,
        opacity: 0,
        duration: 0.8,
        delay: 0.15,
        ease: "power3.out",
      });

      gsap.from(activeRef.current, {
        y: 45,
        opacity: 0,
        scale: 0.96,
        duration: 1,
        delay: 0.2,
        ease: "power3.out",
      });

      gsap.from([previousRef.current, nextRef.current], {
        opacity: 0,
        scale: 0.9,
        duration: 0.9,
        delay: 0.35,
        stagger: 0.1,
        ease: "power3.out",
      });
    }, sectionRef);

    return () => ctx.revert();
  }, []);

  /*
   * CHANGE REVIEW
   */
  const changeReview = (newIndex, newDirection) => {
    if (animating) return;

    setAnimating(true);
    setDirection(newDirection);

    const oldActive = activeRef.current;

    /*
     * OLD ACTIVE CARD
     * moves slightly away and fades
     */
    gsap.to(oldActive, {
      x: newDirection * -60,
      opacity: 0,
      scale: 0.96,
      duration: 0.4,
      ease: "power2.in",
    });

    /*
     * SIDE CARDS
     * slightly fade while preparing for movement
     */
    gsap.to([previousRef.current, nextRef.current], {
      opacity: 0.25,
      duration: 0.25,
      ease: "power2.out",
    });

    /*
     * CHANGE DATA AFTER OLD CARD LEAVES
     */
    setTimeout(() => {
      setActiveIndex(newIndex);
    }, 250);
  };

  /*
   * ANIMATE NEW ACTIVE CARD
   */
  useLayoutEffect(() => {
    if (!animating) return;

    const newActive = activeRef.current;

    gsap.fromTo(
      newActive,
      {
        x: direction * 70,
        opacity: 0,
        scale: 0.94,
      },
      {
        x: 0,
        opacity: 1,
        scale: 1,
        duration: 0.65,
        ease: "power3.out",
      }
    );

    /*
     * New side cards
     */
    gsap.fromTo(
      [previousRef.current, nextRef.current],
      {
        opacity: 0,
        scale: 0.92,
      },
      {
        opacity: 0.5,
        scale: 0.96,
        duration: 0.55,
        delay: 0.05,
        stagger: 0.05,
        ease: "power3.out",
        onComplete: () => {
          setAnimating(false);
        },
      }
    );
  }, [activeIndex]);

  const goNext = () => {
    changeReview(getIndex(activeIndex + 1), 1);
  };

  const goPrevious = () => {
    changeReview(getIndex(activeIndex - 1), -1);
  };

  return (
    <section
      ref={sectionRef}
      className="overflow-hidden bg-[#f7f0e5] py-20 sm:py-24 lg:py-28"
    >
      <div className="mx-auto max-w-[1450px] px-5 sm:px-8 lg:px-10">

        {/*  HEADER  */}

        <div className="flex items-end justify-between">

          <div className="max-w-[620px]">

            <p className="review-heading mb-3 text-[11px] font-medium uppercase tracking-[0.22em] text-[#92251c]">
              Customer Reviews
            </p>

            <h2 className="review-heading text-4xl font-semibold tracking-[-0.045em] text-[#111] sm:text-5xl lg:text-[54px]">
              What Our Guests Say
            </h2>

            <p className="review-description mt-5 max-w-[520px] text-sm leading-6 text-[#666] sm:text-[15px]">
              We take pride in our guests sharing their unforgettable
              experiences. Your feedback inspires us to be even better.
            </p>

          </div>

          {/* DESKTOP ARROWS */}

          <div className="hidden gap-3 sm:flex">

            <button
              onClick={goPrevious}
              disabled={animating}
              className="
                group flex h-11 w-11 items-center justify-center
                rounded-full border border-black/10
                bg-white/70
                transition-all duration-300
                hover:border-[#92251c]/30
                hover:bg-white
                disabled:pointer-events-none
              "
            >
              <ChevronLeft
                size={18}
                className="transition-transform duration-300 group-hover:-translate-x-0.5"
              />
            </button>

            <button
              onClick={goNext}
              disabled={animating}
              className="
                group flex h-11 w-11 items-center justify-center
                rounded-full border border-black/10
                bg-white/70
                transition-all duration-300
                hover:border-[#92251c]/30
                hover:bg-white
                disabled:pointer-events-none
              "
            >
              <ChevronRight
                size={18}
                className="transition-transform duration-300 group-hover:translate-x-0.5"
              />
            </button>

          </div>
        </div>

        {/*  MOBILE ARROWS  */}

        <div className="mt-6 flex gap-3 sm:hidden">

          <button
            onClick={goPrevious}
            disabled={animating}
            className="flex h-10 w-10 items-center justify-center rounded-full border border-black/10 bg-white"
          >
            <ChevronLeft size={17} />
          </button>

          <button
            onClick={goNext}
            disabled={animating}
            className="flex h-10 w-10 items-center justify-center rounded-full border border-black/10 bg-white"
          >
            <ChevronRight size={17} />
          </button>

        </div>

        {/*  CAROUSEL  */}

        <div className="relative mt-14 sm:mt-16">

          <div className="relative h-[390px]">

            {/*  PREVIOUS  */}

            <div
              ref={previousRef}
              className="
                absolute left-[-18%]
                top-1/2
                hidden
                h-[350px]
                w-[38%]
                -translate-y-1/2
                md:block
              "
            >
              <ReviewCard review={reviews[previousIndex]} />
            </div>

            {/*  ACTIVE  */}

            <div
              ref={activeRef}
              className="
                absolute
                left-1/2
                top-0
                z-20
                h-full
                w-full
                -translate-x-1/2
                md:w-[58%]
              "
            >
              <ReviewCard review={reviews[activeIndex]} />
            </div>

            {/*  NEXT  */}

            <div
              ref={nextRef}
              className="
                absolute right-[-18%]
                top-1/2
                hidden
                h-[350px]
                w-[38%]
                -translate-y-1/2
                md:block
              "
            >
              <ReviewCard review={reviews[nextIndex]} />
            </div>

          </div>
        </div>

        {/*  PAGINATION  */}

        <div className="mt-8 flex items-center justify-center gap-2">

          {reviews.map((_, index) => (
            <button
              key={index}
              onClick={() => {
                if (index === activeIndex || animating) return;

                const newDirection =
                  index > activeIndex ? 1 : -1;

                changeReview(index, newDirection);
              }}
              className={`
                h-1.5 rounded-full
                transition-all duration-500
                ${
                  activeIndex === index
                    ? "w-8 bg-[#92251c]"
                    : "w-1.5 bg-[#92251c]/25 hover:bg-[#92251c]/50"
                }
              `}
            />
          ))}

        </div>

      </div>
    </section>
  );
}
